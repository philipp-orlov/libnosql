cmake_minimum_required(VERSION 3.16)
project(nosql_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nosql CONFIG REQUIRED)
get_target_property(nosql_features nosql::nosql INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST nosql_features OR "cxx_std_20" IN_LIST nosql_features)
	message(FATAL_ERROR "Installed nosql must export a C++17 requirement")
endif()
add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE nosql::nosql)
